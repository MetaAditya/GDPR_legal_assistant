=====Pull the code repository==========
      https://github.com/MetaAditya/GDPR_legal_assistant.git

======Ollama Configuration========
   curl -fsSL https://ollama.com/install.sh | sh
   ollama --version
   ollama pull mxbai-embed-large

   cd /etc/systemd/system/ollama.service.d/
   nano override.conf

   sudo systemctl daemon-reload
   sudo systemctl restart ollama

==========Database Configuration==========

   nano /etc/postgresql/18/main/postgresql.conf  -->    listen_addresses = '*'

   sudo docker network inspect legal-assistant-network
   nano /etc/postgresql/18/main/pg_hba.conf ---> 172.16.0.0/12 

   sudo systemctl restart postgresql


   ####Installing PG vector ##############
   Install PGvector-->   https://github.com/pgvector/pgvector
   verify --> sudo -u postgres psql -c "SELECT name, default_version FROM pg_available_extensions WHERE name = 'vector';"

====================Install Apache Age ===================================
   sudo apt update

   sudo apt install -y \
      build-essential \
      git \
      postgresql-server-dev-17 \
      flex \
      bison \
      libreadline-dev \
      zlib1g-dev

   sudo apt install postgresql-server-dev-18

   cd /tmp
   git clone https://github.com/apache/age.git

   cd age
   git checkout release/PG18/1.8.0

   make
   sudo make install

   Verify--> sudo -u postgres psql -c "SELECT name, default_version FROM pg_available_extensions WHERE name = 'age';"

   =================Database Setup===================

   sudo -u postgres psql
   CREATE DATABASE legal_memory_db;
   CREATE DATABASE graph_db_gdpr;

   ALTER USER postgres WITH PASSWORD 'postgres';

   ALTER DATABASE legal_memory_db OWNER TO postgres;
   ALTER DATABASE graph_db_gdpr OWNER TO postgres;

   GRANT ALL PRIVILEGES ON DATABASE legal_memory_db TO postgres;
   GRANT ALL PRIVILEGES ON DATABASE graph_db_gdpr TO postgres;

  
   CREATE EXTENSION IF NOT EXISTS vector;
   CREATE EXTENSION IF NOT EXISTS age;




============Modules Installation===============
   sudo docker network create legal-assistant-network
   sudo docker network ls

   sudo docker pull quay.io/keycloak/keycloak:26.7.1
   sudo docker images

   sudo docker run -d \
   --name keycloak \
   --network legal-assistant-network \
   -p 8080:8080 \

   -e KC_BOOTSTRAP_ADMIN_PASSWORD=admin \
   -e KC_HOSTNAME=http://localhost:8080 \
   quay.io/keycloak/keycloak:26.7.1 \
   start-dev

   sudo docker compose up -d

   sudo docker exec -it gdpr-backend python -c "
